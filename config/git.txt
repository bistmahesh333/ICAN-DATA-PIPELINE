# ==========================================
# Full Git Workflow from Init to Merge
# ==========================================

# Step 0: Navigate to your project folder
# Example:
cd D:\Practice\PySpark\ican-data-pipeline

# Step 1: Initialize a new git repository
git init
# This creates a new local Git repository

# Step 2: Check current branch
git branch
# By default, this is 'master' or 'main'

# Step 3: Create new branches
git branch dev_office   # branch for office environment
git branch dev_home     # branch for home environment

# Step 4: Switch to 'dev_home' to start working
git checkout dev_home
# Now you are on dev_home branch

# Step 5: Make changes to your code
# For example, edit src\main.py to add a simple print:
# print("Hello from dev_home branch!")

# Step 6: Stage your changes
git add .
# Stages all modified and new files

# Step 7: Commit your changes
git commit -m "Initial changes on dev_home branch"

# Step 8: Create remote repository (on GitHub) first
# Copy the HTTPS URL from GitHub, e.g. https://github.com/username/ican-data-pipeline.git

# Step 9: Add remote to local repository
git remote add origin https://github.com/username/ican-data-pipeline.git

# Step 10: Push dev_home branch to remote
git push -u origin dev_home
# -u sets upstream so next time you can just 'git push'

# Step 11: Switch back to main branch
git checkout main

# Step 12: Merge dev_home into main
git merge dev_home
# This brings all dev_home changes into main

# Step 13: Push updated main to remote
git push origin main

# Step 14: Optional - clean up branches if needed
# git branch -d dev_home       # deletes local branch
# git push origin --delete dev_home  # deletes remote branch

# ==========================================
# Notes:
# - Use 'git status' to check changes
# - Use 'git log --oneline --graph --all' to visualize all branches
# - You can follow the same steps for dev_office branch for office changes
# ==========================================
